% ==================== PHẦN 1: BẢN CHẤT MÁY TÍNH VỚI CHỮ ====================
\begin{frame}{1. Câu Hỏi Khởi Đầu: Tại Sao Máy Tính Không Đọc Được Chữ?}
  \begin{columns}[c]
    \begin{column}{0.5\textwidth}
      \begin{block}{Thực tế về máy tính}
        \fontsize{7.5}{9.5}\selectfont
        \begin{itemize}
          \item Máy tính chỉ biết tính toán trên \textbf{con số thực} (cộng, trừ, nhân ma trận).
          \item Đối với máy tính, câu \textit{''Trường Sư Phạm rất đẹp''} chỉ là một mớ ký tự vô tri gồm các mã ASCII/Unicode nhị phân (\texttt{010101...}).
          \item Máy tính không có khái niệm trực giác thế nào là \textit{''đẹp''}, thế nào là \textit{''xấu''}.
        \end{itemize}
      \end{block}

      \begin{exampleblock}{Mục tiêu của NLP (Biểu diễn từ)}
        \fontsize{7.5}{9.5}\selectfont
        Tìm một chiếc \textbf{cầu nối toán học} để biến mỗi từ ngữ thành một dãy số (vector) sao cho: \textit{Các từ có nghĩa tương tự nhau sẽ có các con số gần nhau!}
      \end{exampleblock}
    \end{column}

    \begin{column}{0.5\textwidth}
      \begin{alertblock}{Cách tiếp cận ngây thơ: Đánh số thứ tự (ID)}
        \fontsize{7.5}{9.5}\selectfont
        Giả sử ta đánh số từ điển theo thứ tự:
        \begin{itemize}
          \item \textit{con\_mèo} = $1$
          \item \textit{con\_chó} = $2$
          \item \textit{xe\_tải} = $3$
        \end{itemize}
        \textbf{Cạm bẫy toán học xuất hiện}:
        \begin{itemize}
          \item Máy tính hiểu: $1 + 2 = 3$ $\implies$ \textit{Mèo + Chó = Xe tải} (?!)
          \item Khoảng cách giữa \textit{Mèo} và \textit{Xe tải} ($3 - 1 = 2$) lại lớn hơn khoảng cách \textit{Mèo} và \textit{Chó} ($2 - 1 = 1$). Nếu đánh số ngẫu nhiên, trật tự này hoàn toàn sai lệch!
        \end{itemize}
        $\implies$ \textbf{Đánh số đơn thuần là ngõ cụt!}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% ==================== PHẦN 2: ONE-HOT VÀ TF-IDF ====================
\begin{frame}{2. Cách Làm Cổ Điển: Mã Hóa One-Hot Có Tốt Không?}
  \begin{columns}[c]
    \begin{column}{0.5\textwidth}
      \begin{block}{Ý tưởng One-Hot Encoding}
        \fontsize{7.5}{9.5}\selectfont
        Nếu từ điển có $|V|$ từ, mỗi từ là một dãy có độ dài $|V|$ với đúng một số $1$ ở vị trí của nó:
        \begin{align*}
          \text{giảng\_viên} &= [1, 0, 0, 0, \dots, 0]^T \\
          \text{thầy\_cô} &= [0, 1, 0, 0, \dots, 0]^T \\
          \text{xe\_máy} &= [0, 0, 1, 0, \dots, 0]^T
        \end{align*}
        \begin{itemize}
          \item \textit{Ưu điểm}: Không còn phép cộng vô lý như đánh số $1, 2, 3$.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.5\textwidth}
      \begin{alertblock}{3 Căn bệnh nghiêm trọng của One-Hot}
        \fontsize{7.5}{9.5}\selectfont
        \begin{enumerate}
          \item \textbf{Khổng lồ và lãng phí bộ nhớ}: Từ điển tiếng Việt có $50,000$ từ $\to$ Mỗi từ là vector 50,000 chiều toàn số 0.
          \item \textbf{Căn bệnh ''Mù ngữ nghĩa'' (Tính trực giao)}:
            $$\mathbf{v}_{\text{giảng\_viên}} \cdot \mathbf{v}_{\text{thầy\_cô}} = 0$$
            $$\mathbf{v}_{\text{giảng\_viên}} \cdot \mathbf{v}_{\text{xe\_máy}} = 0$$
            Máy tính thấy \textit{''giảng viên''} xa lạ với \textit{''thầy cô''} y hệt như với \textit{''xe máy''}!
          \item Không thể đo được độ đồng nghĩa hay trái nghĩa.
        \end{enumerate}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Đếm Từ Thông Minh Hơn: TF-IDF (Tần Suất \& Nghịch Đảo)}
  \begin{columns}[c]
    \begin{column}{0.5\textwidth}
      \begin{block}{Nguyên lý trực giác của TF-IDF}
        \fontsize{7.5}{9.5}\selectfont
        \begin{itemize}
          \item \textbf{TF (Term Frequency)}: Từ nào xuất hiện nhiều trong câu đó thì từ đó quan trọng với câu đó.
          \item \textbf{IDF (Inverse Document Frequency)}: Từ nào câu nào cũng có (như \textit{và, thì, là, của, những}) thì từ đó là \textbf{từ rác}, cần bị phạt điểm nặng!
        \end{itemize}
        \vspace{0.1cm}
        \centering
        \color{HCMUEDarkBlue}\textbf{Điểm số = Tần suất trong bài $\times$ Độ quý hiếm}
      \end{block}
    \end{column}

    \begin{column}{0.5\textwidth}
      \begin{alertblock}{Hạn chế cốt tử: Bỏ quên trật tự từ (Bag-of-Words)}
        \fontsize{7.5}{9.5}\selectfont
        Vì chỉ gom từ vào một cái ''túi'' để đếm số lượng, TF-IDF hoàn toàn không hiểu thứ tự:
        \begin{itemize}
          \item Câu A: \textit{''Thầy dạy \textbf{không khó hiểu} chút nào.''} $\to$ Khen (Tích cực).
          \item Câu B: \textit{''Thầy dạy \textbf{khó hiểu, không} thích.''} $\to$ Chê (Tiêu cực).
        \end{itemize}
        Cả hai câu đều chứa các từ giống hệt nhau $\implies$ \textbf{TF-IDF cho ra hai vector gần như trùng nhau!}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% ==================== PHẦN 3: WORD2VEC ====================
\begin{frame}{3. Cuộc Cách Mạng Word2Vec: Từ Ngữ Thành Tọa Độ Không Gian}
  \begin{columns}[c]
    \begin{column}{0.55\textwidth}
      \begin{block}{Ý tưởng thiên tài của Tomas Mikolov (2013)}
        \fontsize{7.5}{9.5}\selectfont
        Dựa trên triết lý ngôn ngữ học của J.R. Firth (1957):\\
        \centering\textit{''Hãy cho tôi biết bạn bè xung quanh bạn, tôi sẽ nói bạn là ai!''}
        \vspace{0.2cm}
        \raggedright
        \begin{itemize}
          \item Thay vì dùng vector 50,000 chiều thưa thớt, ta nén mỗi từ thành một vector nhỏ gồm \textbf{100 con số thực dày đặc} ($\mathbb{R}^{100}$).
          \item 100 con số này đóng vai trò như \textbf{tọa độ GPS} trên một bản đồ ngữ nghĩa đa chiều.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.45\textwidth}
      \begin{exampleblock}{Phép toán vector kỳ diệu}
        \fontsize{7.5}{9.5}\selectfont
        Trong không gian Word2Vec, khoảng cách hình học phản ánh quan hệ đời thực:
        \vspace{-0.2cm}
        $$\mathbf{v}_{\text{Vua}} - \mathbf{v}_{\text{Đàn\_ông}} + \mathbf{v}_{\text{Phụ\_nữ}} \approx \mathbf{v}_{\text{Nữ\_hoàng}}$$
        $$\mathbf{v}_{\text{Hà\_Nội}} - \mathbf{v}_{\text{Việt\_Nam}} + \mathbf{v}_{\text{Pháp}} \approx \mathbf{v}_{\text{Paris}}$$
        \begin{itemize}
          \item \textit{Từ đồng nghĩa}: \textit{thầy} và \textit{giảng viên} nằm sát cạnh nhau.
          \item Góc Cosine giữa chúng tiến gần về 1.0!
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Word2Vec Học Như Thế Nào? Hai Trò Chơi Đoán Chữ}
  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \begin{block}{\small\textbf{Trò chơi 1: CBOW (Đoán từ ở giữa)}}
        \fontsize{7.5}{9.5}\selectfont
        \begin{itemize}
          \item Nhìn vào các từ hàng xóm xung quanh để đoán từ trung tâm bị che mất.
        \end{itemize}
        \vspace{0.1cm}
        \centering
        \fbox{\fontsize{8}{10}\selectfont Hôm nay trời \textbf{[ ? ]} và có nắng nhẹ}
        \vspace{0.1cm}
        \raggedright
        $\to$ Mô hình đoán chữ: \textbf{đẹp} / \textbf{ấm}.\\
        $\to$ Thích hợp học các từ thông dụng, tốc độ huấn luyện rất nhanh.
      \end{block}
    \end{column}

    \begin{column}{0.5\textwidth}
      \begin{block}{\small\textbf{Trò chơi 2: Skip-Gram (Đoán các từ bạn bè)}}
        \fontsize{7.5}{9.5}\selectfont
        \begin{itemize}
          \item Cho một từ trung tâm duy nhất, bắt mô hình đoán các từ thường đi cùng nó.
        \end{itemize}
        \vspace{0.1cm}
        \centering
        \fbox{\fontsize{8}{10}\selectfont Từ trung tâm: \textbf{giảng\_viên}}
        \vspace{0.1cm}
        \raggedright
        $\to$ Mô hình đoán quanh đó có: \textit{nhiệt\_tình, bài\_tập, slide, điểm}.\\
        $\to$ Cực kỳ xuất sắc trong việc học các \textbf{từ hiếm gặp}.
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.25cm}
  \begin{alertblock}{Bí quyết tốc độ: Negative Sampling (SGNS)}
    \fontsize{7.5}{9.5}\selectfont
    Thay vì bắt máy tính so sánh với toàn bộ 50,000 từ trong từ điển (rất chậm), ta chỉ bắt nó phân biệt từ đúng với \textbf{5 từ ngẫu nhiên chọn bừa} (mẫu âm). Máy tính học nhanh hơn hàng trăm lần!
  \end{alertblock}
\end{frame}

% ==================== PHẦN 4: TỪ TỪ ĐẾN CÂU ====================
\begin{frame}{4. Từ Vector Từng Từ Đến Cả Câu: Cạm Bẫy Cộng Trung Bình}
  \begin{columns}[c]
    \begin{column}{0.5\textwidth}
      \begin{block}{Cách làm đơn giản nhất: Average Word2Vec}
        \fontsize{7.5}{9.5}\selectfont
        Lấy các vector của từng từ trong câu rồi tính \textbf{trung bình cộng} lại để thành 1 vector đại diện câu:
        $$\mathbf{v}_{\text{câu}} = \frac{1}{N} (\mathbf{v}_{w_1} + \mathbf{v}_{w_2} + \dots + \mathbf{v}_{w_N})$$
        \begin{itemize}
          \item Dễ cài đặt, chạy cực nhanh, không cần huấn luyện thêm.
          \item Nhưng tại sao trong thực nghiệm, nó lại cho kết quả kém hơn cả TF-IDF?
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.5\textwidth}
      \begin{alertblock}{Cạm bẫy: Nghịch lý triệt tiêu phủ định}
        \fontsize{7.5}{9.5}\selectfont
        \begin{enumerate}
          \item \textbf{Phép cộng có tính giao hoán}: $A + B = B + A$. Câu đảo ngược thứ tự từ vẫn sinh ra vector y hệt nhau!
          \item \textbf{Chữ ''KHÔNG'' bị hòa tan}:
            \begin{itemize}\fontsize{7}{8.5}\selectfont
              \item Từ \textit{''không''} xuất hiện ở mọi nơi, cạnh cả từ khen lẫn chê $\to$ Vector của nó nằm ở chính giữa trung tâm không gian.
              \item Khi lấy $\mathbf{v}_{\text{tốt}} + \mathbf{v}_{\text{không}}$, vector kết quả chỉ hơi lệch một chút so với \textit{''tốt''}.
              \item Phép cộng chỉ \textbf{pha loãng} chứ không thể \textbf{lật ngược 180 độ} cực tính từ Tích cực sang Tiêu cực!
            \end{itemize}
        \end{enumerate}
        $\implies$ \textbf{Cần một mô hình biết đọc tuần tự theo thời gian!}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% ==================== PHẦN 5: MẠNG LSTM ====================
\begin{frame}{5. Mạng Nơ-ron Hồi Quy LSTM: Chiếc Ba Lô Trí Nhớ Thông Minh}
  \begin{columns}[c]
    \begin{column}{0.5\textwidth}
      \begin{block}{Con người đọc câu như thế nào?}
        \fontsize{7.5}{9.5}\selectfont
        Chúng ta đọc từng từ một từ trái sang phải. Khi đọc đến từ thứ 5, trong đầu ta vẫn còn lưu giữ thông tin của 4 từ trước đó.\\
        $\to$ Mạng RNN/LSTM mô phỏng chính xác quá trình này qua \textbf{Trạng thái ẩn (Hidden state $\mathbf{h}_t$)} và \textbf{Ô nhớ dài hạn ($\mathbf{c}_t$)}.
      \end{block}
      \vspace{0.1cm}
      \begin{alertblock}{Tại sao RNN thường bị ''đãng trí''?}
        \fontsize{7.5}{9.5}\selectfont
        RNN đơn giản bị hiện tượng tiêu biến gradient (\textit{Vanishing Gradient}): Câu dài trên 10 từ là nó quên sạch những gì đã đọc ở đầu câu!
      \end{alertblock}
    \end{column}

    \begin{column}{0.5\textwidth}
      \begin{exampleblock}{Giải pháp của LSTM: 3 Cổng gác thông minh}
        \fontsize{7.5}{9.5}\selectfont
        LSTM như một người đi du lịch mang theo chiếc ba lô:
        \begin{enumerate}
          \item \textbf{Cổng quên (Forget Gate)}: Bỏ bớt rác cũ trong ba lô không còn dùng đến.
          \item \textbf{Cổng nạp (Input Gate)}: Chọn lọc những món đồ mới có giá trị nhét vào ba lô.
          \item \textbf{Cổng ra (Output Gate)}: Quyết định từ những thứ trong ba lô, hiện tại nên nói ra điều gì.
        \end{enumerate}
        $\implies$ \textbf{LSTM có thể nhớ ngữ cảnh cách xa hàng chục từ mà không bị quên!}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Mạng BiLSTM: Đọc Xuôi \& Đọc Ngược Cùng Lúc}
  \begin{columns}[c]
    \begin{column}{0.52\textwidth}
      \begin{block}{Tại sao một chiều là chưa đủ?}
        \fontsize{7.5}{9.5}\selectfont
        Hãy thử đoán nghĩa từ \textbf{''độ''} trong câu:
        \begin{center}
          \textit{''Chiếc xe này đã được \textbf{độ} ...''}
        \end{center}
        \begin{itemize}
          \item Nếu chỉ đọc từ trái sang phải, ta chưa rõ là \textit{độ xe} hay \textit{nhiệt độ}.
          \item Nhưng nếu đọc tiếp phần sau: \textit{''... lên công suất 150 mã lực''}, ta hiểu ngay ngữ cảnh!
        \end{itemize}
      \end{block}

      \begin{exampleblock}{BiLSTM (Bidirectional LSTM)}
        \fontsize{7.5}{9.5}\selectfont
        Chạy 2 mạng song song:
        \begin{itemize}
          \item Một mạng đọc từ trái sang phải ($w_1 \to w_T$).
          \item Một mạng đọc từ phải sang trái ($w_T \to w_1$).
          \item Ghép hai góc nhìn lại: $\mathbf{h}_t = [\overrightarrow{\mathbf{h}_t} \,;\, \overleftarrow{\mathbf{h}_t}]$ $\to$ Hiểu trọn vẹn ngữ cảnh hai phía.
        \end{itemize}
      \end{exampleblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Nhưng vẫn còn 1 điểm yếu: Nghẽn cổ chai (Bottleneck)}
        \fontsize{7.5}{9.5}\selectfont
        BiLSTM truyền thống gom toàn bộ câu thành vector bước cuối cùng $\mathbf{h}_T$.\\
        \textbf{Hậu quả}: Giống như bắt bạn đọc cuốn sách 20 trang rồi chỉ được viết đúng 1 dòng tóm tắt. Rất nhiều thông tin chi tiết quý giá ở giữa câu bị rơi rụng!\\
        \vspace{0.2cm}
        $\implies$ \textbf{Làm sao để không bỏ sót bất kỳ từ nào?}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% ==================== PHẦN 6: ATTENTION ====================
\begin{frame}{6. Cơ Chế Chú Ý (Attention Mechanism): Cây Bút Dạ Quang}
  \begin{columns}[c]
    \begin{column}{0.5\textwidth}
      \begin{block}{Ẩn dụ đời thực}
        \fontsize{7.5}{9.5}\selectfont
        Khi đọc một câu nhận xét:
        \begin{center}
          \textit{''Thầy giảng dạy rất \textbf{nhiệt tình} và bài tập \textbf{dễ hiểu}.''}
        \end{center}
        Bộ não chúng ta không ghi nhớ đồng đều từng chữ. Chúng ta cầm một cây \textbf{bút dạ quang} tô sáng vào hai từ: \textbf{nhiệt tình} và \textbf{dễ hiểu}!
      \end{block}

      \begin{exampleblock}{Toán học hóa cây bút dạ quang}
        \fontsize{7.5}{9.5}\selectfont
        Cơ chế Attention tính cho mỗi từ một trọng số $\alpha_t \in [0, 1]$ (tổng các $\alpha = 100\%$):
        $$\mathbf{v}_{\text{toàn câu}} = \sum_{t=1}^T \alpha_t \mathbf{h}_t$$
        Từ nào mang cảm xúc mạnh $\to$ nhận $\alpha_t$ lớn!
      \end{exampleblock}
    \end{column}

    \begin{column}{0.5\textwidth}
      \begin{block}{Bản đồ nhiệt Attention trong thực tế}
        \centering
        \includegraphics[width=\textwidth]{images/attention_sample_pos.png}\\
        {\fontsize{7}{8.5}\selectfont \textit{Hình: Mô hình tự động dồn trọng số vào từ quyết định.}}
      \end{block}
      \begin{alertblock}{Explainable AI (Khả năng giải thích)}
        \fontsize{7.5}{9}\selectfont
        Mô hình không còn là chiếc hộp đen bí ẩn. Nhìn vào trọng số $\alpha_t$, con người biết chính xác \textbf{vì sao máy tính lại dự đoán là Tích cực hay Tiêu cực}!
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% ==================== PHẦN 7: TRANSFORMER & PHOBERT ====================
\begin{frame}{7. Bước Nhảy Vọt Lên Transformer \& PhoBERT Tiếng Việt SOTA}
  \begin{columns}[c]
    \begin{column}{0.52\textwidth}
      \begin{block}{Giới hạn của Word2Vec tĩnh}
        \fontsize{7.5}{9.5}\selectfont
        Word2Vec gán mỗi từ đúng 1 vector cố định:
        \begin{itemize}
          \item \textit{''Tôi uống \textbf{nước} cam''} (chất lỏng).
          \item \textit{''Bảo vệ đất \textbf{nước}''} (quốc gia).
        \end{itemize}
        Cả hai từ \textit{''nước''} trên đều bị ép chung một tọa độ vector $\implies$ Không chuẩn xác!
      \end{block}

      \begin{exampleblock}{Cuộc cách mạng Contextual Embedding}
        \fontsize{7.5}{9.5}\selectfont
        Mô hình \textbf{PhoBERT (VinAI, 2020)} dựa trên Transformer:
        \begin{itemize}
          \item Huấn luyện trên \textbf{20GB báo chí tiếng Việt} (~3 tỷ từ).
          \item Vector của một từ được sinh ra \textbf{động} tùy theo ngữ cảnh của câu cụ thể.
          \item Đọc hiểu cấu trúc câu phức tạp và ẩn ý sâu sắc.
        \end{itemize}
      \end{exampleblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Bí kíp trị Teencode: Tokenizer BPE}
        \fontsize{7.5}{9.5}\selectfont
        Khi gặp từ lóng hoặc viết tắt trên mạng:
        \begin{itemize}
          \item Word2Vec tĩnh: Bó tay, gán thành từ rác \texttt{UNK} (Unknown).
          \item \textbf{PhoBERT (Byte-Pair Encoding)}: Tự động bẻ nhỏ từ lạ thành các mảnh âm tiết quen thuộc để suy luận.
        \end{itemize}
        $\implies$ Giúp PhoBERT đạt độ bền bỉ vượt trội trước ngôn ngữ mạng xã hội đầy nhiễu!
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

    \begin{column}{0.5\textwidth}
      \begin{block}{\small\textbf{1. Biểu diễn từ là nền tảng}}
        \fontsize{7.5}{9.5}\selectfont
        Không có vector chất lượng, không mô hình học sâu nào hoạt động tốt. Từ One-Hot rời rạc $\to$ Word2Vec liên tục là bước ngoặt vĩ đại của NLP.
      \end{block}
      \begin{block}{\small\textbf{2. Trật tự câu quyết định ý nghĩa}}
        \fontsize{7.5}{9.5}\selectfont
        Đừng chỉ cộng trung bình các từ. Hãy dùng mạng xử lý chuỗi như \textbf{LSTM/BiLSTM} để giữ trọn ngữ cảnh xuôi - ngược.
      \end{block}
      \begin{block}{\small\textbf{3. Cơ chế Chú ý mở toang hộp đen}}
        \fontsize{7.5}{9.5}\selectfont
        \textbf{Self-Attention} không chỉ tăng độ chính xác mà còn cho ta biết lý do tại sao AI đưa ra quyết định đó (\textbf{Explainable AI}).
      \end{block}
    \end{column}

    \begin{column}{0.5\textwidth}
      \begin{block}{\small\textbf{4. Transformer là tương lai}}
        \fontsize{7.5}{9.5}\selectfont
        Các mô hình tiền huấn luyện lớn như \textbf{PhoBERT} giải quyết bài toán đa nghĩa và bền bỉ trước tiếng lóng nhờ cơ chế Tokenizer BPE.
      \end{block}
      \begin{exampleblock}{\small\textbf{5. Hiểu dữ liệu quan trọng hơn chỉnh mô hình}}
        \fontsize{7.5}{9.5}\selectfont
        Mô hình đạt 95\% ở miền giáo dục có thể tụt xuống 72\% ở miền TMĐT nếu không hiểu về sự dịch chuyển phân phối (\textbf{Domain Gap}).
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}
