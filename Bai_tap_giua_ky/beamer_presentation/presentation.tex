\begin{frame}{1. Đặt Vấn Đề: Thách Thức Biểu Diễn Ngữ Nghĩa Tiếng Việt}
  \begin{columns}[T, onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Thách thức cốt lõi của Biểu diễn từ}
        \fontsize{8}{10.5}\selectfont
        \begin{itemize}\setlength{\itemsep}{0.25em}
          \item Ánh xạ ngôn ngữ rời rạc sang vector dày đặc $\mathbb{R}^d$ bảo toàn \textbf{khoảng cách ngữ nghĩa}.
          \item \textbf{Đặc thù tiếng Việt}: Ngôn ngữ đơn lập, không biến hình, từ ghép, từ láy, đa nghĩa, đảo ngữ, cấu trúc phủ định và sắc thái châm biếm (\textit{sarcasm}).
          \item \textbf{Giả thuyết phân bố (Firth, 1957)}: \textit{''You shall know a word by the company it keeps''} — một từ được định nghĩa bởi các từ xuất hiện xung quanh nó.
        \end{itemize}
      \end{block}
    \end{column}
    \hfill
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Hạn chế của mô hình cổ điển (One-Hot, TF-IDF)}
        \fontsize{8}{10.5}\selectfont
        \begin{itemize}\setlength{\itemsep}{0.25em}
          \item \textbf{Bùng nổ số chiều}: Kích thước ma trận bằng $|V|$ ($10^4 - 10^5$), cực kỳ lãng phí bộ nhớ.
          \item \textbf{Tính trực giao tuyệt đối}: $\mathbf{e}_i^T \mathbf{e}_j = 0, \forall i \ne j$. Không đo được sự tương đồng giữa \textit{''giảng viên''} và \textit{''thầy cô''}.
          \item \textbf{Mất trật tự từ (Bag-of-Words)}: Bỏ qua hoàn toàn cú pháp và phụ thuộc ngữ cảnh dài hạn.
        \end{itemize}
      \end{alertblock}

\begin{frame}{Không Gian Vector Word2Vec: CBOW, Skip-Gram \& SGNS}
  \begin{columns}[T, onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{CBOW vs Skip-Gram (Mục Tiêu Tối Ưu)}
        \fontsize{7.5}{9.5}\selectfont
        \textbf{CBOW}: Dự đoán từ tâm $w_t$ từ ngữ cảnh $w_{t\pm c}$:
        $$\displaystyle \mathcal{L}_{\text{CBOW}} = \sum_{t=1}^T \log P(w_t \mid w_{t-c}, \dots, w_{t+c})$$
        \textbf{Skip-Gram}: Từ từ tâm $w_t$, dự đoán ngữ cảnh:
        $$\displaystyle \mathcal{L}_{\text{SG}} = \sum_{t=1}^T \sum_{-c \le j \le c, j \ne 0} \log P(w_{t+j} \mid w_t)$$
        \textit{Skip-Gram học từ hiếm và ngữ nghĩa đa dạng rất tốt.}
      \end{block}
    \end{column}
    \hfill
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Negative Sampling (SGNS) Tối Ưu Tốc Độ}
        \fontsize{7.5}{9.5}\selectfont
        \textbf{Vấn đề Softmax}: Mẫu số $\sum_{w' \in V} \exp(\mathbf{v}'^T_{w'} \mathbf{v}_w)$ tốn $O(|V|)$.\\
        \textbf{Giải pháp SGNS}: Phân loại nhị phân giữa cặp thật $(w_I, w_O)$ và $K$ mẫu âm ($K \in [5, 20]$):
        $$\displaystyle \mathcal{L}_{\text{SGNS}} = \log \sigma(\mathbf{v}'^T_{w_O} \mathbf{v}_{w_I}) + \sum_{k=1}^K \mathbb{E}_{w_{n_k}} \left[ \log \sigma(-\mathbf{v}'^T_{w_{n_k}} \mathbf{v}_{w_I}) \right]$$
        \begin{itemize}\setlength{\itemsep}{0.15em}
          \item Giảm độ phức tạp từ $O(|V|) \to O(K)$.
          \item Phân phối unigram mũ $3/4$: $P_n(w) \propto f(w)^{3/4}$.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{So Sánh Đối Đầu Kỹ Thuật: CBOW vs Skip-Gram}
  \fontsize{8}{10}\selectfont
  Bảng tổng hợp đối chiếu trực tiếp các tiêu chí cốt lõi giữa hai kiến trúc Word2Vec kinh điển:

  \vspace{0.18cm}
  \resizebox{\textwidth}{!}{
  \begin{tabular}{p{3.0cm}p{5.8cm}p{5.8cm}}
    \toprule
    \textbf{Tiêu chí so sánh} & \textbf{CBOW (Continuous Bag-of-Words)} & \textbf{Skip-Gram (với Negative Sampling)} \\
    \midrule
    \textbf{Cơ chế dự đoán} & Dự đoán từ trung tâm $w_t$ dựa vào ngữ cảnh & Dự đoán các từ ngữ cảnh $w_{t+j}$ từ từ tâm $w_t$ \\
    \textbf{Tốc độ huấn luyện} & \textbf{Rất nhanh} (gấp nhiều lần Skip-Gram) & Chậm hơn do xét từng cặp từ trong cửa sổ \\
    \textbf{Khả năng học từ hiếm} & Trung bình (bị lấn át bởi các từ phổ biến) & \textbf{Xuất sắc} (học hình học vector rất kỹ cho từ hiếm) \\
    \textbf{Độ phức tạp tính toán} & $O(C \times d + d \times \log|V|)$ & $O(C \times (d + d \times K))$ với SGNS ($K$ mẫu âm) \\
    \textbf{Ngữ liệu phù hợp} & Kho văn bản lớn, ưu tiên tốc độ hội tụ & Dữ liệu vừa/nhỏ, yêu cầu ngữ nghĩa từ hiếm sâu \\
    \bottomrule
  \end{tabular}
  }

  \vspace{0.25cm}
  \begin{block}{\small\textbf{Lựa chọn thực nghiệm trong đề tài}}
    \fontsize{7.5}{9.5}\selectfont
    Đề tài sử dụng \textbf{Word2Vec CBOW} ($d=100$, window=5, min\_count=1, 30 epochs) để tạo pretrained embedding, đồng thời đối chuẩn độc lập với Skip-Gram để kiểm chứng tính bền vững của không gian vector ngữ nghĩa.
  \end{block}
\end{frame}

\begin{frame}{Học Biểu Diễn Văn Bản Bằng LSTM \& Mạng BiLSTM}
  \begin{columns}[T, onlytextwidth]
    \begin{column}{0.50\textwidth}
      \begin{block}{Cơ chế kiểm soát của LSTM (3 Cổng)}
        \fontsize{7.5}{9.5}\selectfont
        Khắc phục suy biến Gradient bằng \textbf{Ô nhớ $\mathbf{c}_t$} và 3 cổng:
        \begin{enumerate}\setlength{\itemsep}{0.1em}
          \item \textbf{Cổng quên}: $\mathbf{f}_t = \sigma(W_f [\mathbf{h}_{t-1}, \mathbf{x}_t] + \mathbf{b}_f)$
          \item \textbf{Cổng nạp}: $\mathbf{i}_t = \sigma(W_i [\mathbf{h}_{t-1}, \mathbf{x}_t] + \mathbf{b}_i)$
          \item \textbf{Ứng viên nhớ}: $\tilde{\mathbf{c}}_t = \tanh(W_c [\mathbf{h}_{t-1}, \mathbf{x}_t] + \mathbf{b}_c)$
          \item \textbf{Cập nhật ô nhớ}: $\mathbf{c}_t = \mathbf{f}_t \odot \mathbf{c}_{t-1} + \mathbf{i}_t \odot \tilde{\mathbf{c}}_t$
          \item \textbf{Cổng ra}: $\mathbf{o}_t = \sigma(W_o [\mathbf{h}_{t-1}, \mathbf{x}_t] + \mathbf{b}_o)$
          \item \textbf{Trạng thái ẩn}: $\mathbf{h}_t = \mathbf{o}_t \odot \tanh(\mathbf{c}_t)$
        \end{enumerate}
      \end{block}
    \end{column}
    \hfill
    \begin{column}{0.46\textwidth}
      \begin{block}{Mạng hồi quy hai chiều BiLSTM}
        \fontsize{7.5}{9.5}\selectfont
        Xử lý văn bản theo cả hai hướng đồng thời:
        \begin{itemize}\setlength{\itemsep}{0.1em}
          \item Chiều xuôi: $\overrightarrow{\mathbf{h}_t} = \text{LSTM}_{fwd}(\mathbf{x}_t, \overrightarrow{\mathbf{h}_{t-1}})$
          \item Chiều ngược: $\overleftarrow{\mathbf{h}_t} = \text{LSTM}_{bwd}(\mathbf{x}_t, \overleftarrow{\mathbf{h}_{t+1}})$
          \item Kết hợp: $\mathbf{h}_t = [\overrightarrow{\mathbf{h}_t} \,;\, \overleftarrow{\mathbf{h}_t}] \in \mathbb{R}^{2 \cdot d_{hidden}}$
        \end{itemize}
      \end{block}
      \vspace{0.15cm}
      \begin{alertblock}{Điểm nghẽn thông tin (Bottleneck)}
        \fontsize{7.5}{9.5}\selectfont
        BiLSTM thông thường lấy vector cuối: $\mathbf{v}_D = [\overrightarrow{\mathbf{h}_T} \,;\, \overleftarrow{\mathbf{h}_1}]$.\\
        $\to$ Câu dài bị nén qua một vector duy nhất $\implies$ suy giảm tín hiệu đầu câu!
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Đề Xuất: Tích Hợp Additive Self-Attention \& Khả Giải Thích XAI}
  \begin{columns}[T, onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Cơ chế Chú ý Tự thân (Additive Attention)}
        \fontsize{7.5}{9.5}\selectfont
        Tính \textbf{tổng có trọng số} của toàn bộ chuỗi trạng thái ẩn $\mathbf{h}_t$, không bỏ rơi bất kỳ từ nào:
        \begin{align*}
          \mathbf{u}_t &= \tanh(\mathbf{W}_a \mathbf{h}_t + \mathbf{b}_a) \\
          \displaystyle \alpha_t &= \frac{\exp(\mathbf{u}_t^T \mathbf{u}_w)}{\sum_{\tau=1}^T \exp(\mathbf{u}_\tau^T \mathbf{u}_w)}, \quad \mathbf{v}_D = \sum_{t=1}^T \alpha_t \mathbf{h}_t
        \end{align*}
        \begin{itemize}\setlength{\itemsep}{0.15em}
          \item $\mathbf{u}_w$: Vector ngữ cảnh học được (\textit{Context query}).
          \item $\alpha_t \in [0, 1]$ với $\sum_t \alpha_t = 1$: Trọng số đóng góp token.
        \end{itemize}
      \end{block}
    \end{column}
    \hfill
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Giá trị Explainable AI (XAI) trực quan}
        \fontsize{7.5}{9.5}\selectfont

\begin{frame}{Thảo Luận Học Thuật: Nghịch Lý Word2Vec \& Transfer Learning}
  \begin{columns}[T, onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{1. Vì sao Avg Word2Vec kém hơn TF-IDF?}
        \fontsize{7.5}{9.5}\selectfont
        \begin{itemize}\setlength{\itemsep}{0.2em}
          \item \textbf{Mất trật tự từ tính giao hoán}: Phép cộng $\frac{1}{|D|}\sum \mathbf{v}_w$ biến \textit{''không khó hiểu''} và \textit{''khó hiểu không''} thành cùng vector!
          \item \textbf{Nghịch lý triệt tiêu phủ định}: Từ \textit{''không''} ở tâm không gian; phép cộng chỉ \textbf{pha loãng} chứ không thể \textbf{đảo chiều} cực tính vector.
          \item \textbf{TF-IDF giữ độc lập}: Logistic Regression gán trọng số âm lớn cho từ phủ định rất hiệu quả.
        \end{itemize}
      \end{alertblock}
    \end{column}
    \hfill
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{2. Giá trị Transfer Learning Pretrained W2V}
        \fontsize{7.5}{9.5}\selectfont
        \begin{itemize}\setlength{\itemsep}{0.2em}
          \item \textbf{Bứt phá độ chính xác}: BiLSTM Pretrained tăng \textbf{+1.83\%} so với khởi tạo ngẫu nhiên Scratch.
          \item \textbf{Giải phóng năng lực học}: Mạng không phải học hình học từ đầu; toàn lực tập trung vào \textbf{học lan truyền ngữ cảnh theo thời gian}.
          \item \textbf{Hội tụ nhanh $2\times$}: Hạn chế tối đa hiện tượng quá khớp (overfitting) trên tập dữ liệu nhỏ.
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Thảo Luận Học Thuật: Bứt Phá Attention \& Thách Thức Domain Gap}
  \begin{columns}[T, onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{3. Cơ chế giúp Attention bứt phá?}
        \fontsize{7.5}{9.5}\selectfont
        \begin{itemize}\setlength{\itemsep}{0.2em}
          \item \textbf{Bứt phá +5.33\% trên E-Commerce}: Tăng từ 67.17\% lên \textbf{72.50\%}.
          \item \textbf{Xóa bỏ Information Bottleneck}: Không ép câu qua $\mathbf{h}_T$; tạo đường tắt kết nối trực tiếp mọi token tới bộ phân loại.
          \item \textbf{Soft Feature Selection}: Tự động hạ thấp trọng số hư từ, khuếch đại tính từ đối lập trong câu phức.
        \end{itemize}
      \end{block}
    \end{column}
    \hfill
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{4. Nguyên nhân sụt giảm khi sang TMĐT?}
        \fontsize{7.5}{9.5}\selectfont
        \begin{itemize}\setlength{\itemsep}{0.2em}
          \item \textbf{Bùng nổ Out-of-Vocabulary (OOV)}: Teencode (\textit{sp, auth, ship}), viết tắt tùy tiện khiến mô hình tĩnh tê liệt.
          \item \textbf{Châm biếm / Sarcasm}: Dùng từ tích cực bề mặt để chê bai chất lượng sản phẩm.
          \item \textbf{PhoBERT bền bỉ nhất} ($\Delta = 9.33\%$): Nhờ cơ chế tách subwords BPE bẻ nhỏ từ lạ thành đơn vị con.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Phân Tích Lỗi Định Tính \& Giải Pháp Kỹ Thuật Đề Xuất}
  \begin{columns}[T, onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Ca 1: Cấu trúc câu tương phản đảo hướng}
        \fontsize{7.5}{9.5}\selectfont
        \begin{itemize}\setlength{\itemsep}{0.15em}
          \item \textbf{Thực tế}: \textit{''Giao hàng hơi lâu nhưng chất lượng áo thì đỉnh chóp.''} $\to$ \textbf{TÍCH CỰC}.
          \item \textbf{BiLSTM dự đoán}: Tiêu cực (Lỗi OOV từ lóng và trượt theo vế \textit{''hơi lâu''}).
        \end{itemize}
        \vspace{0.08cm}
        \textbf{\color{HCMUEDarkBlue}$\implies$ Giải pháp kỹ thuật khắc phục}:
        \begin{enumerate}\fontsize{7}{8.5}\selectfont\setlength{\itemsep}{0.1em}
          \item \textbf{Phân rã liên từ nghịch tiếp}: Tách vế sau \textit{''nhưng''} và tăng trọng số phạt (Contrastive Loss Weighting).
          \item \textbf{Bổ sung từ điển Teencode}: Chuẩn hóa tự động \textit{''đỉnh chóp''} $\to$ \textit{''rất tốt''}.
        \end{enumerate}
      \end{block}
    \end{column}
    \hfill
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Ca 2: Ngôn ngữ mỉa mai / châm biếm (Sarcasm)}
        \fontsize{7.5}{9.5}\selectfont
        \begin{itemize}\setlength{\itemsep}{0.15em}
          \item \textbf{Thực tế}: \textit{''Shop phục vụ quá nhiệt tình, nhắn tin 3 ngày mới thèm rep.''} $\to$ \textbf{TIÊU CỰC}.
          \item \textbf{Cả BiLSTM và PhoBERT}: Đoán Tích cực (Bị đánh lừa bởi cụm \textit{''quá nhiệt tình''}).
        \end{itemize}
        \vspace{0.08cm}
        \textbf{\color{HCMUERed}$\implies$ Giải pháp kỹ thuật khắc phục}:
        \begin{enumerate}\fontsize{7}{8.5}\selectfont\setlength{\itemsep}{0.1em}
          \item \textbf{Tích hợp Đồ thị Tri thức (Knowledge Graph)}: Bổ sung tri thức đời thực: \textit{''chờ 3 ngày''} là sự cố trễ hạn.
          \item \textbf{Suy luận Chuỗi tư duy (Chain-of-Thought)}: Fine-tune với reasoning steps kiểm tra tính tương phản hành vi.
        \end{enumerate}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}
